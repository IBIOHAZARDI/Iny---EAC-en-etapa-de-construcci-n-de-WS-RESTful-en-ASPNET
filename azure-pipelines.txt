variables:
  buildConfiguration: Debug
  testProject: SecurityAsserts/SecurityAsserts.csproj
  testResultsDir: $(Build.ArtifactStagingDirectory)\test-results
  sonarQubeServiceConnection: SonarQube
  sonarProjectKey: Inyector-SecurityAsserts
  sonarProjectName: Inyector SecurityAsserts
  vulnerableApiUrl: http://localhost:5000
  patchedApiUrl: http://localhost:5002
  zapTargetUrl: http://host.docker.internal:5000
  zapImage: zaproxy/zap-stable:2.16.1
  zapFailOnAlert: false
  iisDeployEnabled: 'false'
  iisVulnerableApiPackagePath: ''
  iisPatchedApiPackagePath: ''
  iisVulnerableApiPhysicalPath: 'C:\inetpub\wwwroot\VulnerableApi'
  iisPatchedApiPhysicalPath: 'C:\inetpub\wwwroot\VulnerableApi_Patched'
  iisVulnerableApiAppPool: VulnerableApi
  iisPatchedApiAppPool: VulnerableApi_Patched

pool:
  name: Default

stages:
  - stage: Build
    displayName: Compilacion
    jobs:
      - job: BuildSecurityAsserts
        displayName: Restaurar y compilar SecurityAsserts
        steps:
          - task: UseDotNet@2
            displayName: Usar .NET SDK 8.0.319
            inputs:
              packageType: sdk
              version: 8.0.319

          - task: SonarQubePrepare@7
            displayName: Preparar analisis SonarQube
            inputs:
              SonarQube: $(sonarQubeServiceConnection)
              scannerMode: MSBuild
              projectKey: $(sonarProjectKey)
              projectName: $(sonarProjectName)
              extraProperties: |
                sonar.exclusions=**/bin/**,**/obj/**,**/TestResults/**

          - task: DotNetCoreCLI@2
            displayName: Restaurar dependencias
            inputs:
              command: restore
              projects: $(testProject)

          - task: DotNetCoreCLI@2
            displayName: Compilar SecurityAsserts
            inputs:
              command: build
              projects: $(testProject)
              arguments: --configuration $(buildConfiguration) --no-restore

  - stage: SonarQube
    displayName: Analisis estatico y quality gate
    dependsOn: Build
    jobs:
      - job: Analyze
        displayName: Analisis SonarQube
        steps:
          - task: UseDotNet@2
            displayName: Usar .NET SDK 8.0.319
            inputs:
              packageType: sdk
              version: 8.0.319

          - task: SonarQubePrepare@7
            displayName: Preparar analisis SonarQube
            inputs:
              SonarQube: $(sonarQubeServiceConnection)
              scannerMode: MSBuild
              projectKey: $(sonarProjectKey)
              projectName: $(sonarProjectName)
              extraProperties: |
                sonar.exclusions=**/bin/**,**/obj/**,**/TestResults/**

          - task: DotNetCoreCLI@2
            displayName: Restaurar dependencias
            inputs:
              command: restore
              projects: $(testProject)

          - task: DotNetCoreCLI@2
            displayName: Compilar para SonarQube
            inputs:
              command: build
              projects: $(testProject)
              arguments: --configuration $(buildConfiguration) --no-restore

          - task: SonarQubeAnalyze@7
            displayName: Analizar codigo con SonarQube

          - task: SonarQubePublish@7
            displayName: Publicar quality gate de SonarQube
            inputs:
              pollingTimeoutSec: '300'

  - stage: DeployIIS
    displayName: Despliegue opcional en IIS
    dependsOn: SonarQube
    condition: and(succeeded(), eq(variables['iisDeployEnabled'], 'true'))
    jobs:
      - job: DeployApis
        displayName: Publicar APIs en IIS local
        steps:
          - powershell: |
              $ErrorActionPreference = 'Stop'
              Import-Module WebAdministration

              $apis = @(
                @{
                  Name = 'VulnerableApi'
                  PackagePath = '$(iisVulnerableApiPackagePath)'
                  PhysicalPath = '$(iisVulnerableApiPhysicalPath)'
                  AppPool = '$(iisVulnerableApiAppPool)'
                },
                @{
                  Name = 'VulnerableApi_Patched'
                  PackagePath = '$(iisPatchedApiPackagePath)'
                  PhysicalPath = '$(iisPatchedApiPhysicalPath)'
                  AppPool = '$(iisPatchedApiAppPool)'
                }
              )

              foreach ($api in $apis) {
                if ([string]::IsNullOrWhiteSpace($api.PackagePath) -or -not (Test-Path -LiteralPath $api.PackagePath)) {
                  throw "No existe el paquete de $($api.Name): '$($api.PackagePath)'"
                }
                if (-not (Test-Path "IIS:\AppPools\$($api.AppPool)")) {
                  throw "No existe el application pool '$($api.AppPool)' en IIS."
                }

                New-Item -ItemType Directory -Force -Path $api.PhysicalPath | Out-Null
                $poolWasStarted = (Get-WebAppPoolState -Name $api.AppPool).Value -eq 'Started'
                if ($poolWasStarted) {
                  Stop-WebAppPool -Name $api.AppPool
                }

                try {
                  Get-ChildItem -LiteralPath $api.PackagePath -Force | ForEach-Object {
                    Copy-Item -LiteralPath $_.FullName -Destination $api.PhysicalPath -Recurse -Force
                  }
                } finally {
                  if ($poolWasStarted) {
                    Start-WebAppPool -Name $api.AppPool
                  }
                }
              }
            displayName: Desplegar paquetes en application pools IIS

  - stage: SecurityValidation
    displayName: SecurityAsserts y gates A/B
    dependsOn:
      - Build
      - SonarQube
      - DeployIIS
    condition: and(in(dependencies.Build.result, 'Succeeded'), in(dependencies.SonarQube.result, 'Succeeded'), in(dependencies.DeployIIS.result, 'Succeeded', 'Skipped'))
    jobs:
      - job: RunSecurityAsserts
        displayName: Ejecutar escenarios A y B
        steps:
          - task: UseDotNet@2
            displayName: Usar .NET SDK 8.0.319
            inputs:
              packageType: sdk
              version: 8.0.319

          - powershell: |
              $ErrorActionPreference = 'Stop'
              New-Item -ItemType Directory -Force -Path '$(testResultsDir)' | Out-Null

              $scenarios = @(
                @{ Name = 'ScenarioA'; ApiName = 'VulnerableApi'; Url = '$(vulnerableApiUrl)' },
                @{ Name = 'ScenarioB'; ApiName = 'VulnerableApi_Patched'; Url = '$(patchedApiUrl)' }
              )

              foreach ($scenario in $scenarios) {
                try {
                  $response = Invoke-WebRequest "$($scenario.Url)/health" -UseBasicParsing -TimeoutSec 10
                  if ($response.StatusCode -ne 200) {
                    throw "HTTP $($response.StatusCode)"
                  }
                } catch {
                  throw "$($scenario.ApiName) no esta disponible en $($scenario.Url): $($_.Exception.Message)"
                }
              }

              foreach ($scenario in $scenarios) {
                $env:TEST_API_URL = $scenario.Url
                $trxFile = "$($scenario.Name)_$(Build.BuildId).trx"
                & dotnet test '$(testProject)' --configuration '$(buildConfiguration)' `
                  --logger "trx;LogFileName=$trxFile" `
                  --results-directory '$(testResultsDir)'
                if ($LASTEXITCODE -ne 0) {
                  Write-Warning "$($scenario.Name) detecto controles no satisfechos (exit=$LASTEXITCODE); el gate evaluara el TRX."
                }
              }
              Remove-Item Env:\TEST_API_URL -ErrorAction SilentlyContinue
            displayName: Ejecutar asserts en ambos escenarios
            continueOnError: true

          - powershell: |
              $ErrorActionPreference = 'Stop'

              function Get-TrnFromTrx([string]$path, [string]$scenarioName) {
                if (-not (Test-Path -LiteralPath $path)) {
                  throw "No se genero TRX para $scenarioName: $path"
                }

                [xml]$trxXml = Get-Content -Raw -LiteralPath $path
                $counters = $trxXml.TestRun.ResultSummary.Counters
                $passed = [double]$counters.passed
                $total = [double]$counters.total
                if ($total -eq 0) {
                  throw "El TRX de $scenarioName no contiene resultados validos."
                }

                return ($passed / $total) * 100.0
              }

              $scenarioAPath = Join-Path '$(testResultsDir)' 'ScenarioA_$(Build.BuildId).trx'
              $scenarioBPath = Join-Path '$(testResultsDir)' 'ScenarioB_$(Build.BuildId).trx'
              $trnA = Get-TrnFromTrx $scenarioAPath 'Escenario A'
              $trnB = Get-TrnFromTrx $scenarioBPath 'Escenario B'

              Write-Host "TRN Escenario A (vulnerable): $([math]::Round($trnA, 2))%"
              Write-Host "TRN Escenario B (parcheado): $([math]::Round($trnB, 2))%"

              if ($trnA -ge 95) {
                throw "Quality gate A fallido: TRN = $([math]::Round($trnA, 2))% >= 95%. No se confirma la deteccion de la API vulnerable."
              }
              if ($trnB -le 95) {
                throw "Quality gate B fallido: TRN = $([math]::Round($trnB, 2))% <= 95%. Se requiere un valor estrictamente superior al 95%."
              }

              Write-Host 'Quality gates aprobados: A < 95% y B > 95%.'
            displayName: Validar TRN esperada en A y B

          - task: PublishTestResults@2
            displayName: Publicar resultados TRX
            condition: always()
            inputs:
              testResultsFormat: VSTest
              testResultsFiles: $(testResultsDir)\*.trx
              failTaskOnFailedTests: false
              failTaskOnMissingResultsFile: false
              testRunTitle: SecurityAsserts

          - task: PublishPipelineArtifact@1
            displayName: Guardar TRX de SecurityAsserts
            condition: always()
            inputs:
              targetPath: $(testResultsDir)
              artifact: security-test-results
              publishLocation: pipeline

  - stage: DynamicAnalysis
    displayName: Analisis dinamico OWASP ZAP
    dependsOn:
      - DeployIIS
      - SecurityValidation
    condition: and(in(dependencies.DeployIIS.result, 'Succeeded', 'Skipped'), in(dependencies.SecurityValidation.result, 'Succeeded', 'SucceededWithIssues', 'Failed'))
    jobs:
      - job: RunZap
        displayName: Escanear VulnerableApi
        steps:
          - powershell: |
              $ErrorActionPreference = 'Stop'
              $zapOutput = Join-Path '$(Build.ArtifactStagingDirectory)' 'zap-results'
              New-Item -ItemType Directory -Force -Path $zapOutput | Out-Null

              docker pull '$(zapImage)'
              docker run --rm `
                -v "${zapOutput}:/zap/wrk/:rw" `
                '$(zapImage)' `
                zap-baseline.py -t '$(zapTargetUrl)' -r zap-report.html -J zap-report.json -I
              $zapExitCode = $LASTEXITCODE

              if ($zapExitCode -ne 0) {
                Write-Warning "OWASP ZAP finalizo con codigo $zapExitCode. Revise el reporte publicado."
                if ('$(zapFailOnAlert)' -eq 'true') {
                  throw "ZAP bloqueo el pipeline (exit=$zapExitCode)."
                }
              }
            displayName: Analizar VulnerableApi con ZAP 2.16.1
            continueOnError: false

          - task: PublishPipelineArtifact@1
            displayName: Guardar reportes ZAP
            condition: always()
            inputs:
              targetPath: $(Build.ArtifactStagingDirectory)\zap-results
              artifact: zap-security-scan
              publishLocation: pipeline

  - stage: PublishEvidence
    displayName: Publicar evidencias consolidadas
    dependsOn:
      - SecurityValidation
      - DynamicAnalysis
    condition: and(always(), in(dependencies.SecurityValidation.result, 'Succeeded', 'SucceededWithIssues', 'Failed'), in(dependencies.DynamicAnalysis.result, 'Succeeded', 'SucceededWithIssues', 'Failed'))
    jobs:
      - job: ConsolidateEvidence
        displayName: Consolidar resultados de seguridad
        steps:
          - task: DownloadPipelineArtifact@2
            displayName: Descargar TRX
            inputs:
              artifact: security-test-results
              path: $(Pipeline.Workspace)\security-test-results

          - task: DownloadPipelineArtifact@2
            displayName: Descargar reportes ZAP
            inputs:
              artifact: zap-security-scan
              path: $(Pipeline.Workspace)\zap-security-scan

          - powershell: |

              $evidencePath = Join-Path '$(Build.ArtifactStagingDirectory)' 'security-evidence'
              New-Item -ItemType Directory -Force -Path $evidencePath | Out-Null
              Copy-Item -Path '$(Pipeline.Workspace)\security-test-results\*' -Destination $evidencePath -Recurse -Force
              Copy-Item -Path '$(Pipeline.Workspace)\zap-security-scan\*' -Destination $evidencePath -Recurse -Force
            displayName: Preparar paquete de evidencias

          - task: PublishPipelineArtifact@1
            displayName: Publicar paquete de evidencias
            inputs:
              targetPath: $(Build.ArtifactStagingDirectory)\security-evidence
              artifact: security-analysis
              publishLocation: pipeline
